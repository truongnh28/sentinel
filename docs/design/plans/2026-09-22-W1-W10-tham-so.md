**Mục tiêu.** Mỗi tham số của AuditGame-SE trả lời được câu hỏi *"vì sao là số này?"* bằng một phép đo, một suy dẫn, hoặc một lập luận kèm kiểm chứng — thay cho giá trị gán tay hiện tại.

```python
#!/usr/bin/env python3
"""measure_kappa.py -- bảng kappa đo bằng CPU-time, từ chính vòng chạy game.

Không dựng bàn đo riêng: chạy episode thật với record_traces=True rồi gộp
TaskTrace.audit_seconds theo "stage:carrier". Đo cái đang chạy, không đo một
bản sao của nó.

    python3 tools/measure_kappa.py --reps 30 > spikes/kappa-cpu.json
"""
from __future__ import annotations
import argparse, json, random, statistics, subprocess, datetime, sys, pathlib

sys.path.insert(0, str(pathlib.Path(__file__).resolve().parent.parent))

import agent, build, detector, runner
import policies as P
from core import CARRIERS, seed_of

STAGES = ("insertion", "retrieval", "delegation", "commit")


def measure(reps: int = 30, h: int = 8, setting: str = "mid") -> dict:
    det = detector.Detector.from_setting(setting)
    ag = agent.MockAgent()
    per_stage: dict = {s: [] for s in STAGES}
    for i in range(reps):
        wf = build.make_workflow(f"kappa-{i:03d}", "django", h,
                                 random.Random(seed_of("kappa", i)))
        ps = build.plan_poison(wf, "memory", 2, random.Random(seed_of("kappa-ps", i)))
        if ps is None:
            continue
        pol = P.make_policy("B2 uniform random", 17.95, seed_of("kappa-pol", i), setting)
        res = runner.run_once(wf, ps, pol, det, ag, seed=i, record_traces=True)
        for tr in res.traces:
            for key, secs in tr.audit_seconds.items():
                stage = key.split(":")[0]
                if stage in per_stage:
                    per_stage[stage].append(secs)
    kappa = {s: statistics.median(v) for s, v in per_stage.items() if v}
    spread = {s: (min(v), max(v), len(v)) for s, v in per_stage.items() if v}
    vals = list(kappa.values())
    chi = (max(vals) - min(vals)) / (sum(vals) / len(vals)) if vals else 0.0
    return {
        "kappa_cpu_seconds": kappa,
        "spread_min_max_n": spread,
        "chi": chi,
        "chi_formula": "max|k_i - k_j| / mean(k)",
        "provenance": {
            "clock": "time.process_time (user+sys)",
            "reps": reps, "H": h, "setting": setting,
            "date": datetime.date.today().isoformat(),
            "commit": subprocess.run(["git", "rev-parse", "HEAD"],
                                     capture_output=True, text=True).stdout.strip(),
        },
    }


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--reps", type=int, default=30)
    print(json.dumps(measure(ap.parse_args().reps), indent=2, ensure_ascii=False))
```
